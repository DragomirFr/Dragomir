def click_at(x: int, y: int, delay: float = 0.4):
    """Click specific coordinates."""
    pyautogui.click(x, y)
    time.sleep(delay)


def copy_then_click_and_paste(x: int, y: int, text: str, delay: float = 0.4):
    """Pre-load text to clipboard, focus field, clear it, and paste."""
    pyperclip.copy(text)
    time.sleep(0.1)

    pyautogui.click(x, y)
    time.sleep(0.15)

    mod = "command" if sys.platform == "darwin" else "ctrl"
    pyautogui.hotkey(mod, "a")
    pyautogui.press("backspace")
    time.sleep(0.1)

    pyautogui.hotkey(mod, "v")
    time.sleep(delay)


def enter_otp_digits(x: int, y: int, digits: str, delay: float = 0.25):
    """Focuses the OTP input and fires discrete key events for each digit."""
    pyautogui.click(x, y)
    time.sleep(0.3)

    for digit in str(digits).strip():
        pyautogui.press(digit)
        time.sleep(delay)
    time.sleep(0.3)


def get_email_body(msg: Message) -> str:
    """Extract readable text/HTML content from an email."""
    parts = []
    if msg.is_multipart():
        for part in msg.walk():
            ctype = part.get_content_type()
            if ctype in ("text/plain", "text/html"):
                p = part.get_payload(decode=True)
                if isinstance(p, bytes):
                    parts.append(p.decode("utf-8", errors="ignore"))
    else:
        p = msg.get_payload(decode=True)
        if isinstance(p, bytes):
            parts.append(p.decode("utf-8", errors="ignore"))
    return "\n".join(parts)


def extract_code(html: str) -> str | None:
    """Look for 4-digit code matching pattern in email body."""
    soup = BeautifulSoup(html, "html.parser")
    text = soup.get_text(" ", strip=True)
    match = re.search(r"Your Verification Code:\s*(\d{4})", text, re.IGNORECASE)
    return match.group(1) if match else None


def get_existing_email_ids(mail: imaplib.IMAP4) -> set[bytes]:
    """Snapshot all currently existing UNSEEN email IDs."""
    status, messages = mail.search(None, "UNSEEN")
    if status == "OK" and messages and messages[0]:
        return set(messages[0].split())
    return set()


def wait_for_verification_code(
    mail: imaplib.IMAP4,
    ignored_ids: set[bytes],
    timeout: int = 120,
    poll_interval: float = 0.5,
) -> str:
    """Polls INBOX, inspecting only newly arrived emails, newest first."""
    print("[*] Waiting for new verification email via Proton Bridge...")
    start = time.time()

    while time.time() - start < timeout:
        mail.noop()
        status, messages = mail.search(None, "UNSEEN")
        if status == "OK" and messages and messages[0]:
            all_ids = messages[0].split()
            new_ids = [eid for eid in reversed(all_ids) if eid not in ignored_ids]

            for email_id in new_ids:
                status, data = mail.fetch(email_id, "(RFC822)")
                if status != "OK":
                    continue

                raw_email = None
                for item in data:
                    if isinstance(item, tuple) and len(item) >= 2 and isinstance(item[1], bytes):
                        raw_email = item[1]
                        break

                if raw_email is None:
                    continue

                msg = email.message_from_bytes(raw_email)
                code = extract_code(get_email_body(msg))
                if code:
                    mail.store(email_id, "+FLAGS", "\\Seen")
                    print(f"[+] Retrieved code: {code}")
                    return code

        time.sleep(poll_interval)

    raise TimeoutError("Timed out waiting for new verification code.")
